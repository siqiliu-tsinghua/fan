
式~\eqref{eq:7-4-84}
又诱导式~\eqref{eq:7-4-43}
中的
$\widehat{\mathfrak g}^{\circ}$。

\begin{proposition}{7.4.85}\label{prop:7-4-85}
$\widehat{\mathfrak g}^{\circ}$
是一个 $L_\infty$ 同态。
\end{proposition}

\begin{proof}
需要核对下式：
\begin{equation}
\begin{aligned}
&\sum_i
(-1)^{*'(i;\mathbf Y)}
\mathfrak g^{\circ}
\bigl(
Y_1,\ldots,\delta(Y_i),\ldots,Y_\ell
\bigr)(\mathbf x)\\
&\quad+
\sum_{i<j}
(-1)^{*'(i,j;\mathbf Y)}
\mathfrak g^{\circ}
\bigl(
[Y_i,Y_j],Y_1,\ldots,
\widehat Y_i,\ldots,
\widehat Y_j,\ldots,Y_\ell
\bigr)(\mathbf x)\\
&=
\sum_c
\sum_{\mathbf Y_{(1)}
\cup\mathbf Y_{(2)}=\mathbf Y}
(-1)^{
*'(\mathbf Y_{(1)},\mathbf Y_{(2)},c;\mathbf Y)
}\\[-0.4ex]
&\qquad\cdot
\mathfrak g^{\circ}
\bigl(\mathbf Y_{(1)}\bigr)
\bigl(
\mathbf x_c^{3;1},
\mathfrak g^{\circ}
\bigl(\mathbf Y_{(2)}\bigr)
(\mathbf x_c^{3;2}),
\mathbf x_c^{3;3}
\bigr).
\end{aligned}
\tag{7.4.86}\label{eq:7-4-86}
\end{equation}
这里的符号指数定义为
\[
\begin{aligned}
*'(i;\mathbf Y)
&=*(i)+\deg Y_i-1
=\deg'Y_1+\cdots+\deg'Y_i-1,\\
*'(i,j;\mathbf Y)
&=*(i,j)-\deg'Y_i
+\deg Y_i(\deg Y_j+1),\\
*'(\mathbf Y_{(1)},\mathbf Y_{(2)},c;\mathbf Y)
&=*(\mathbf Y_{(1)},\mathbf Y_{(2)},c;\mathbf Y)
-\deg'\mathbf Y_{(1)}\\
&\quad+
(\deg'\mathbf Y_{(1)}+1)\deg'\mathbf Y_{(2)}\\
&\equiv
*(\mathbf Y_{(1)},\mathbf Y_{(2)},c;\mathbf Y)
+\deg'\mathbf Y\\
&\quad+
\deg'\mathbf Y_{(1)}
\deg'\mathbf Y_{(2)}
\pmod 2.
\end{aligned}
\]
上式中的
$*(i)$、
$*(i,j)$ 与
$*(\mathbf Y_{(1)},\mathbf Y_{(2)},c;\mathbf Y)$
按式~\eqref{eq:7-4-57}
规定。
第一行中的
$\deg Y_i$
来自 $\delta$ 与
$\iota_1$ 的符号差；
原符号 $*(i)$ 中的 $+1$
不出现在 $*'(i;\mathbf Y)$ 中，
故将其减去。
第二行中的
$\deg Y_i(\deg Y_j+1)$
来自括号 $[\ ,\ ]$
与 $\iota_2$ 的符号差；
原符号 $*(i,j)$ 中的
$\deg'Y_i$
不出现在
$*'(i,j;\mathbf Y)$ 中，
故从第二行右端减去。
第三行中，
原符号
$*(\mathbf Y_{(1)},\mathbf Y_{(2)},c;\mathbf Y)$
里的 $+1$
与式~\eqref{eq:7-4-28}
中的负号抵消，
故不出现在该行右端。

除符号外，
式~\eqref{eq:7-4-86}
由式~\eqref{eq:7-4-57}
得到。
下面核对符号。
令
\[
(Y_1,\ldots,Y_\ell)
=(Z_\ell,\ldots,Z_1)
=\operatorname{Op}(Z_1,\ldots,Z_\ell),
\]
并将式~\eqref{eq:7-4-57}
应用于
$(Y_1,\ldots,Y_\ell)$。
左端第一项成为
\begin{equation}
\begin{aligned}
&\sum_i
(-1)^{
\deg'Y_1+\cdots+\deg'Y_{i-1}+1
}
\mathfrak g
\bigl(
Y_1,\ldots,\delta(Y_i),
\ldots,Y_\ell
\bigr)(\mathbf x)\\
&=
\sum_i
(-1)^{
\deg'Z_1+\cdots+\deg'Z_i
+\deg'\mathbf Z+1
}
\mathfrak g^{\circ}
\bigl(
Z_1,\ldots,\delta(Z_i),
\ldots,Z_\ell
\bigr)(\mathbf x)\\
&=
\sum_i
(-1)^{
*'(i;\mathbf Z)+\deg'\mathbf Z
}
\mathfrak g^{\circ}
\bigl(
Z_1,\ldots,\delta(Z_i),
\ldots,Z_\ell
\bigr)(\mathbf x).
\end{aligned}
\tag{7.4.87}\label{eq:7-4-87}
\end{equation}

式~\eqref{eq:7-4-57}
左端第二项变为
\[
\begin{aligned}
&\sum_{i<j}
(-1)^{*(i,j)}
\mathfrak g\bigl(
[Y_i,Y_j],Y_1,\ldots,
\widehat Y_i,\ldots,
\widehat Y_j,\ldots,Y_\ell
\bigr)(\mathbf x)\\
&=
\sum_{i<j}
(-1)^{*(i,j)}
\mathfrak g^{\circ}
\bigl(
Y_\ell,\ldots,
\widehat Y_j,\ldots,
\widehat Y_i,\ldots,
Y_1,[Y_i,Y_j]
\bigr)(\mathbf x)\\
&=
\sum_{i<j}
(-1)^{
*(i,j)+*_1(i,j)
}
\mathfrak g^{\circ}
\bigl(
[Y_i,Y_j],
Y_\ell,\ldots,
\widehat Y_j,\ldots,
\widehat Y_i,\ldots,Y_1
\bigr)(\mathbf x)\\
&=
\sum_{i<j}
(-1)^{
*(\ell-j+1,\ell-i+1)
+*_1(\ell-j+1,\ell-i+1)
+*_2(i,j)
}\\
&\qquad\cdot
\mathfrak g^{\circ}
\bigl(
[Z_i,Z_j],
Z_1,\ldots,
\widehat Z_i,\ldots,
\widehat Z_j,\ldots,Z_\ell
\bigr)(\mathbf x).
\end{aligned}
\]
这里
$*(i,j)$
按式~\eqref{eq:7-4-57}
规定，且
\[
\begin{aligned}
*_1(i,j)
&=
(\deg'\mathbf Y
-\deg'Y_i-\deg'Y_j)
(\deg'Y_i+\deg'Y_j+1),\\
*_2(i,j)
&=\deg Z_i\deg Z_j+1.
\end{aligned}
\]
于是
\[
\begin{aligned}
&*(\ell-j+1,\ell-i+1)
+*_1(\ell-j+1,\ell-i+1)
+*_2(i,j)\\
&=
\deg'Y_{\ell-j+1}
\sum_{k<\ell-j+1}\deg'Y_k
+\deg'Y_{\ell-i+1}
\sum_{\substack{k<\ell-i\\
k\ne\ell-j+1}}
\deg'Y_{k+1}
\\
&\quad+\deg'Y_{\ell-j+1}\\
&\quad+
(\deg'\mathbf Z-\deg'Z_i-\deg'Z_j)
(\deg'Z_i+\deg'Z_j+1)
+\deg Z_i\deg Z_j+1\\
&=
\deg'Z_i
\sum_{\substack{k>i\\k\ne j}}\deg'Z_k
+\deg'Z_j
\sum_{k>j}\deg'Z_k\\
&\quad+\deg'Z_j\\
&\quad+
(\deg'\mathbf Z-\deg'Z_i-\deg'Z_j)
(\deg'Z_i+\deg'Z_j+1)
+\deg Z_i\deg Z_j+1\\
&\equiv
\deg'Z_i
\sum_{k<i}\deg'Z_k
+\deg'Z_j
\sum_{\substack{k<j\\k\ne i}}
\deg'Z_k
\\
&\quad+\deg Z_i(\deg Z_j+1)
+\deg'\mathbf Z
\pmod2.
\end{aligned}
\]
即
\begin{equation}
\begin{aligned}
*'(i,j;\mathbf Z)
&=
*(\ell-j+1,\ell-i+1)\\
&\quad+
*_1(\ell-j+1,\ell-i+1)
+*_2(i,j)+\deg'\mathbf Z.
\end{aligned}
\tag{7.4.88}\label{eq:7-4-88}
\end{equation}

再令 $\operatorname{Shuff}(\mathbf Y)$
表示 $\mathbf Y$ 的所有洗牌
$(\mathbf Y_{(1)},\mathbf Y_{(2)})$
所成之集。
显然，
\[
\operatorname{Shuff}(\mathbf Z)
=
\bigl\{
\bigl(
\operatorname{Op}(\mathbf Y_{(1)}),
\operatorname{Op}(\mathbf Y_{(2)})
\bigr)
\mid
(\mathbf Y_{(1)},\mathbf Y_{(2)})
\in\operatorname{Shuff}(\mathbf Y)
\bigr\}.
\]
还有下式：
\begin{equation}
\begin{aligned}
&*\bigl(
\operatorname{Op}(\mathbf Y_{(1)}),
\operatorname{Op}(\mathbf Y_{(2)}),
c;\mathbf Z
\bigr)\\
&\qquad\equiv
*(\mathbf Y_{(1)},\mathbf Y_{(2)},c;\mathbf Y)
+\deg'\mathbf Y_{(1)}
\deg'\mathbf Y_{(2)}\\
&\qquad\equiv
*'(\mathbf Y_{(1)},\mathbf Y_{(2)},c;\mathbf Y)
+\deg'\mathbf Z
\pmod2.
\end{aligned}
\tag{7.4.89}\label{eq:7-4-89}
\end{equation}
由式~\eqref{eq:7-4-87}、
\eqref{eq:7-4-88}
及~\eqref{eq:7-4-89}，
命题~\ref{prop:7-4-85}
由命题~\ref{prop:7-4-56}
推出。
\end{proof}
