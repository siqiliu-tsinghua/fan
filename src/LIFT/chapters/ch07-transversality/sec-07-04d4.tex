
它在符号暂略时对应于元素
\[
Z(e_0)\bigl(
Z(v_b)\bigl(
Z(e_1)(x_1),
Z(e_4)\bigl(
Z(v_a)\bigl(
Z(e_2)(x_2),Z(e_3)(x_3)
\bigr)\bigr)\bigr)\bigr).
\]
将所有算子移至左侧、
所有 $x_i$ 移至右侧，
得到次序
\[
Z(e_0),Z(v_b),Z(e_1),
Z(e_4),Z(v_a),Z(e_2),Z(e_3),
x_1,x_2,x_3.
\]
此移动产生的符号为
\[
(-1)^{
\deg x_1
\bigl(
\deg Z(e_4)+\deg Z(v_a)
+\deg Z(e_2)+\deg Z(e_3)
\bigr)
+\deg x_2\deg Z(e_3)
}.
\]
因此，图~\ref{fig:7-4-5}
给出
\[
\begin{aligned}
&\mathfrak h(\Gamma)(x_1,x_2,x_3)\\
&\quad=(-1)^{
\deg x_1
\bigl(
\deg Z(e_4)+\deg Z(v_a)
+\deg Z(e_2)+\deg Z(e_3)
\bigr)
+\deg x_2\deg Z(e_3)
}\\
&\qquad\cdot
Z(e_0)\bigl(
Z(v_b)\bigl(
Z(e_1)(x_1),
Z(e_4)\bigl(
Z(v_a)\bigl(
Z(e_2)(x_2),Z(e_3)(x_3)
\bigr)\bigr)\bigr)\bigr).
\end{aligned}
\]

\begin{remark}{7.4.60}\label{rem:7-4-60}
实际上，
$\mathfrak h(\Gamma)$
的构造推广了
$\mathfrak f(\Gamma)$ 与
$\mathfrak g(\Gamma)$
的构造。
下面说明原因。
取 $\Gamma\in G(k,\mathbf Y)$，
为其配置两套装饰
$Z_\Gamma^{\mathfrak f}$、
$Z_\Gamma^{\mathfrak g}$\translatornote{源书此处把两套逐边、逐点定义的装饰直接写成 $\operatorname{Hom}(B_kD,D)$ 中的算子；依前页定义，属于该同态空间的是复合后的 $\mathfrak h(|\Gamma|,Z_\Gamma)$。见疑误 A-0573。}：
若 $e_0$ 是第 $0$ 个外顶点
所邻接的边，置
\[
Z_\Gamma^{\mathfrak f}(e_0)=\mathfrak G,
\qquad
Z_\Gamma^{\mathfrak g}(e_0)=\Pi_H.
\]
对其余所有外边 $e_i$，置
\[
Z_\Gamma^{\mathfrak f}(e_i)
=Z_\Gamma^{\mathfrak g}(e_i)
=\operatorname{id}.
\]
对任意内边 $e$，置
\[
Z_\Gamma^{\mathfrak f}(e)
=Z_\Gamma^{\mathfrak g}(e)
=\mathfrak G.
\]
若 $v$ 为 $Y$ 型顶点，置
\[
Z_\Gamma^{\mathfrak f}(v)
=Z_\Gamma^{\mathfrak g}(v)
=Y(v).
\]
若 $v$ 为邻接 $k_v+1$
条边的 $M$ 型顶点，置
\[
Z_\Gamma^{\mathfrak f}(v)
=Z_\Gamma^{\mathfrak g}(v)
=\mathfrak m_{k_v,\beta(v)}.
\]
于是，由定义可见，
除符号外，
\begin{equation}
\left\{
\begin{aligned}
\mathfrak h(|\Gamma|,Z_\Gamma^{\mathfrak f})
&=(-1)^{*(\Gamma)}\mathfrak f(\Gamma),\\
\mathfrak h(|\Gamma|,Z_\Gamma^{\mathfrak g})
&=(-1)^{*(\Gamma)}\mathfrak g(\Gamma).
\end{aligned}
\right.
\tag{7.4.61}\label{eq:7-4-61}
\end{equation}
这里的 $\mathfrak f(\Gamma)$ 与
$\mathfrak g(\Gamma)$
如式~\eqref{eq:7-4-54} 所示\translatornote{源书如此回指；但 (7.4.54) 是对树求和，单树的 $\mathfrak f(\Gamma),\mathfrak g(\Gamma)$ 由 (7.4.49.1)--(7.4.52) 递归给出。见疑误 A-0574。}。
在证明命题~\ref{prop:7-4-56}
时，我们将通过式~\eqref{eq:7-4-68}
给出式~\eqref{eq:7-4-61}
中的符号。
\end{remark}

下面的引理~7.4.64
描述 $\mathfrak h(\Gamma)$
的微分。
为陈述它，
还需引入一些记号。
令 $\Gamma=(|\Gamma|,Z)$。
对每条边 $e$，
用下式定义 $\partial_eZ$：
\begin{equation}
\left\{
\begin{aligned}
\partial_eZ(v)&=Z(v),
&&\text{$v$ 为顶点},\\
\partial_eZ(e')&=Z(e'),
&&e'\ne e,\\
\partial_eZ(e)
&=\delta\circ Z(e)
-(-1)^{\deg Z(e)}
Z(e)\circ\widehat\delta.
\end{aligned}
\right.
\tag{7.4.62}\label{eq:7-4-62}
\end{equation}
对每个内顶点 $v$，
用下式定义 $\partial_vZ$：
\begin{equation}
\left\{
\begin{aligned}
\partial_vZ(e)&=Z(e),
&&\text{$e$ 为边},\\
\partial_vZ(v')&=Z(v'),
&&v'\ne v,\\
\partial_vZ(v)
&=\delta\circ Z(v)
-(-1)^{\deg Z(v)}
Z(v)\circ\widehat\delta.
\end{aligned}
\right.
\tag{7.4.63}\label{eq:7-4-63}
\end{equation}
这里
$\widehat\delta\colon BD\to BD$
由
$\delta\colon D\to D$
按式~\eqref{eq:3-2-1}
定义；
它不同于式~\eqref{eq:7-4-27}
中的 Hochschild 微分
$\delta$。

\begin{lemma}{7.4.64}\label{lem:7-4-64}
有
\[
\begin{aligned}
&(\delta\circ
\mathfrak h(|\Gamma|,Z))(\mathbf x)
-(-1)^{\deg\mathfrak h(|\Gamma|,Z)}
\bigl(
\mathfrak h(|\Gamma|,Z)
\circ\widehat\delta
\bigr)(\mathbf x)\\
&\qquad=
\sum_e(-1)^{*(e)}
\mathfrak h(|\Gamma|,\partial_eZ)
(\mathbf x)
+\sum_v(-1)^{*(v)}
\mathfrak h(|\Gamma|,\partial_vZ)
(\mathbf x).
\end{aligned}
\]
\end{lemma}

为描述这里的
$*(e)$ 与 $*(v)$，
作如下定义。

\begin{definition}{7.4.65}\label{def:7-4-65}
设 $|\Gamma|$ 是上述树。
若顶点（或边）$f'$
位于顶点（或边）$f$ 的左侧，
记为
$f'\mid_{\mathrm{left}}f$。
其含义由以下条件确定。

取
$f,f'\in C^0(\Gamma)\cup C^1(\Gamma)$\translatornote{源书此处把顶点限定为 $C^0_{\mathrm{ext}}(\Gamma)$；但下文式 (7.4.67) 对内顶点 $v$ 也使用该关系，故译文采用全体顶点。见疑误 A-0571。}。
以 $l(f)$ 表示 $|\Gamma|$
中连接 $f$ 与第 $0$ 个外顶点
$v_0$ 的最小路径\liftIndex{I343}。
\begin{enumerate}
\item[\textup{(7.4.66.1)}]\label{cond:7-4-66-1}
若 $f'\subset l(f)$
且 $f'\ne f$，
则 $f'\mid_{\mathrm{left}}f$。

\item[\textup{(7.4.66.2)}]\label{cond:7-4-66-2}
若 $f\subset l(f')$，
则 $f'\mid_{\mathrm{left}}f$
不成立。

\item[\textup{(7.4.66.3)}]\label{cond:7-4-66-3}
设 $f\not\subset l(f')$
且 $f'\not\subset l(f)$。
考虑子树
$l(f')\cup l(f)$。
它有三个顶点和一条边\translatornote{源书确实写“三个顶点、一条边”，但这不能构成所述连通子树，亦与图 7.4.6 不合；此处保留源句，判据依下一句的循环次序。见疑误 A-0572。}。
当且仅当它形如
图~\ref{fig:7-4-6}
时，称
$f'\mid_{\mathrm{left}}f$。
亦即，
$(v_0,f,f')$
遵循平面逆时针定向。
\end{enumerate}
\end{definition}

现在置
\begin{equation}
*(e)
=\sum_{f\mid_{\mathrm{left}}e}
\deg Z(f),
\qquad
*(v)
=\sum_{f\mid_{\mathrm{left}}v}
\deg Z(f).
\tag{7.4.67}\label{eq:7-4-67}
\end{equation}
接下来，
定义在式~\eqref{eq:7-4-54}
中暂缓给出的整数
$*(\Gamma)$。
设 $v'_a$ 为满足
$Y(v'_a)=Y_a$ 的 $Y$ 型顶点。
置
\begin{equation}
*(\Gamma)
=\sum_{\substack{
(a,b):\,1\le a<b\le\ell\\
v'_b\mid_{\mathrm{left}}v'_a
}}
\deg'Y_a\,\deg'Y_b.
\tag{7.4.68}\label{eq:7-4-68}
\end{equation}
