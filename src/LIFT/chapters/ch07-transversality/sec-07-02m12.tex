
令 $\mathfrak C_{a,r}(L'_1,\lambda'_1)$
为 $[0,1]_r\times C(L'_1,\lambda'_1)$
的一个模型。
更确切地说，
我们取第 4.2.1 小节中的模型
$C(L'_1,\lambda'_1)^{[0,1]_r}$。
于是存在映射
\[
\mathcal F_a^{s=s_0}\colon
C^{s=s_0}(L'_0,(\lambda_0,\lambda'_0))
\longrightarrow
\mathfrak C_{a,r}(L'_1,\lambda'_1)
\]
满足
\begin{equation}
\operatorname{Eval}_{r=0}\circ\mathcal F_a^{s=s_0}
=j\circ\mathfrak f^{\phi'}\circ j_{s=s_0},
\tag{7.2.306.1}\label{eq:7-2-306-1}
\end{equation}
\begin{equation}
\operatorname{Eval}_{r=1}\circ\mathcal F_a^{s=s_0}
=j\circ j_{s=s_0}\circ
\mathfrak f^{\phi'}_{\mathrm{ext},s=s_0},
\tag{7.2.306.2}\label{eq:7-2-306-2}
\end{equation}
并且在 $C(L'_0,\lambda'_0)$ 上有
\begin{equation}
\mathcal F_a^{s=s_0}
=\operatorname{Incl}_r\circ j\circ\mathfrak f^{\phi'}.
\tag{7.2.306.3}\label{eq:7-2-306-3}
\end{equation}
图式~\ref{diag:7-2-17} 的三个包含箭头反向后，
所得图式在同伦意义下交换，
由此可得 $\mathcal F_a^{s=s_0}$ 的存在性。
原图式在 $C(L'_0,\lambda'_0)$ 上严格交换，
故 (7.2.306.3) 成立。
由 (7.2.306.1) 与 (7.2.306.2) 可知，
\[
\mathcal F_a^{s=s_0}\circ
\mathfrak f^{\psi_{0,s_0}}\colon

\mathfrak C_{a,r}(L'_1,\lambda'_1)
\]

现在取
$[0,1]^2\times C(L'_1,\lambda'_1)$

$\widehat{\mathfrak C}_a(L'_1,\lambda'_1)$，
使其 $s=s_0$ 截面为
$\mathfrak C_{a,r}(L'_1,\lambda'_1)$，
而 $r=r_0$ 截面为
$C([0,1]\times L'_1,\lambda'_1)$。
更确切地说，我们取
\begin{equation}
\widehat{\mathfrak C}_a(L'_1,\lambda'_1)
=C([0,1]_s\times L'_1,\lambda'_1)^{[0,1]_r}.
\tag{7.2.307}\label{eq:7-2-307}
\end{equation}
特别地，存在包含映射
\[
\operatorname{Incl}_r\colon
C([0,1]_s\times L'_1,\lambda'_1)
\longrightarrow
\widehat{\mathfrak C}_a(L'_1,\lambda'_1)
\]
满足
\begin{equation}
\begin{gathered}
\operatorname{Eval}_{s=s_0}\circ\operatorname{Incl}_r
=\operatorname{Incl}_r\circ\operatorname{Eval}_{s=s_0}\\
\colon C([0,1]_s\times L'_1,\lambda'_1)
\longrightarrow
\mathfrak C_{a,r}(L'_1,\lambda'_1).
\end{gathered}
\tag{7.2.308}\label{eq:7-2-308}
\end{equation}

\begin{remark}{7.2.309}\label{rem:7-2-309}
须注意，
对一般的 $[0,1]^2\times C$ 模型，
并不假定存在满足 (7.2.308) 的
$\operatorname{Incl}_r$、
$\operatorname{Incl}_s$
（见注~\ref{rem:7-2-203}）。
这里 (7.2.308) 成立，
是因为我们在 (7.2.307) 中
纯粹代数地由
$C([0,1]_s\times L'_1,\lambda'_1)$
定义了
$\widehat{\mathfrak C}_a(L'_1,\lambda'_1)$。
\end{remark}

\begin{lemma}{7.2.310}\label{lem:7-2-310}
对任一满足 (7.2.306) 的
$\mathcal F_a^{s=s_0}$，
存在滤过 $A_{n,K}$ 同态
\[
\mathfrak H_a\colon
C([0,1]_s\times L'_0,(\lambda_0,\lambda'_0))
\longrightarrow
\widehat{\mathfrak C}_a(L'_1,\lambda'_1)
\]
满足
\begin{equation}
\operatorname{Eval}_{r=0}\circ\mathfrak H_a
=j_{[0,1]_s}\circ
\mathfrak f^{\operatorname{id}\times\phi'}
\circ\mathfrak e_a,
\tag{7.2.311.1}\label{eq:7-2-311-1}
\end{equation}
\begin{equation}
\operatorname{Eval}_{r=1}\circ\mathfrak H_a
=j_{[0,1]_s}\circ\mathfrak E_a\circ
\mathfrak f^{\operatorname{id}\times\phi'}_{\mathrm{ext}},
\tag{7.2.311.2}\label{eq:7-2-311-2}
\end{equation}
\begin{equation}
\operatorname{Eval}_{s=s_0}\circ\mathfrak H_a
=\mathcal F_a^{s=s_0}
\circ\operatorname{Eval}_{s=s_0},
\tag{7.2.311.3}\label{eq:7-2-311-3}
\end{equation}
并且在
$C([0,1]_s\times L'_0,\lambda'_0)$
上有
\begin{equation}
\mathfrak H_a
=\operatorname{Incl}_r\circ j_{[0,1]_s}
\circ\mathfrak f^{\operatorname{id}\times\phi'}.
\tag{7.2.311.4}\label{eq:7-2-311-4}
\end{equation}
\end{lemma}
